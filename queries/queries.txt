CS 513 Team 117 - Chicago Food Inspections
Profiling and integrity-constraint queries, run against raw.db (D) and clean.db (D'),
both built by scripts/join/build_databases.py from one reproducible build process.
D is one flat table (inspection_raw); D' is normalized per the Phase-I ER diagram
(establishment / inspection / violation_finding / violation_code) - the schemas are
not identical, so each pair below is two queries that express the same analytical
question against the appropriate schema in each database, not one shared query.
Actual result values for every query below are captured in notes/S4_before_after_results.md.

================================================================================
Q1 - Failure rate by facility type and year
================================================================================
-- On D (raw.db) - flat table, no join, but facility_type is uncanonicalized
-- text so this over-fragments into ~500 groups instead of one per real category.
SELECT facility_type,
       CAST(substr(inspection_date, 7, 4) AS INTEGER) AS yr,
       COUNT(*) AS inspections,
       AVG(CASE WHEN results = 'Fail' THEN 1.0 ELSE 0 END) AS fail_rate
FROM inspection_raw
WHERE results IN ('Pass', 'Pass w/ Conditions', 'Fail')
GROUP BY facility_type, yr
ORDER BY fail_rate DESC;

-- On D' (clean.db) - canonical facility_type via the establishment join,
-- outcome_bucket instead of raw Results so "no_outcome" rows are excluded.
SELECT e.facility_type,
       CAST(substr(i.inspection_date, 7, 4) AS INTEGER) AS yr,
       COUNT(*) AS inspections,
       AVG(CASE WHEN i.outcome_bucket = 'fail' THEN 1.0 ELSE 0 END) AS fail_rate
FROM inspection i
JOIN establishment e ON e.license_no = i.license_no
WHERE i.outcome_bucket IN ('pass', 'fail')
GROUP BY e.facility_type, yr
ORDER BY fail_rate DESC;

================================================================================
Q2 - Most common violations among failed inspections
================================================================================
-- On D (raw.db) - Violations is one unparsed text blob per row, so there is no
-- row to GROUP BY per violation. The closest thing you *can* run is grouping on
-- the whole blob, which just counts exact-duplicate inspections and answers a
-- different (useless) question. Included to show it quantitatively, not as a
-- real substitute for Q2.
SELECT violations, COUNT(*) AS n
FROM inspection_raw
WHERE results = 'Fail' AND violations != ''
GROUP BY violations
ORDER BY n DESC
LIMIT 15;

-- On D' (clean.db) - real answer, using the era-remapped violation_code so
-- pre/post-07/01/2018 code collisions (Problem 4) don't get merged.
SELECT vc.violation_code, vc.description, COUNT(*) AS times_cited
FROM violation_finding vf
JOIN inspection i ON i.inspection_id = vf.inspection_id
JOIN violation_code vc ON vc.violation_code = vf.violation_code
WHERE i.outcome_bucket = 'fail'
GROUP BY vc.violation_code, vc.description
ORDER BY times_cited DESC
LIMIT 15;

================================================================================
IC1 - License # must identify a real establishment (Problem 3)
================================================================================
-- D: rows with a placeholder license (can't be joined to any facility)
SELECT COUNT(*) FROM inspection_raw WHERE license_no = '0' OR license_no = '';

-- D': same check, after attempting recovery via name+address matching
SELECT COUNT(*) FROM inspection WHERE license_no = '0' OR license_no = '';

-- D': breakdown of how those rows were resolved
SELECT license_fix_status, COUNT(*) FROM inspection GROUP BY license_fix_status;

================================================================================
IC2 - Facility Type must be consistent per establishment (Problem 1)
================================================================================
-- D: license numbers whose inspections disagree on Facility Type
SELECT COUNT(*) FROM (
    SELECT license_no FROM inspection_raw
    WHERE license_no != '0' AND license_no != ''
    GROUP BY license_no
    HAVING COUNT(DISTINCT facility_type) > 1
);

-- D': by construction, establishment has exactly one facility_type per
-- license_no (chosen as the mode across that license's inspections), so this
-- denial constraint can't be violated post-cleaning.
SELECT COUNT(*) FROM (
    SELECT license_no FROM establishment
    GROUP BY license_no
    HAVING COUNT(DISTINCT facility_type) > 1
);

================================================================================
IC4 - Results must be a real pass/fail outcome for U1's fail-rate calculation
================================================================================
-- D: rows where Results doesn't represent an actual inspection outcome, but
-- would still get counted in a naive fail-rate denominator if you forgot to
-- filter (Out of Business, No Entry, Not Ready, Business Not Located)
SELECT results, COUNT(*) FROM inspection_raw
WHERE results NOT IN ('Pass', 'Fail', 'Pass w/ Conditions')
GROUP BY results;

-- D': same rows, now explicitly bucketed so U1 queries can filter them out
-- with outcome_bucket = 'no_outcome' instead of relying on remembering all
-- four raw string values. Queries the actual clean.db inspection table (the
-- earlier version of this query aliased raw_result to itself from
-- inspection_raw and didn't touch outcome_bucket at all - fixed).
SELECT raw_result, outcome_bucket, COUNT(*) AS n_occurrences
FROM inspection
GROUP BY raw_result, outcome_bucket
ORDER BY raw_result, outcome_bucket;

-- D': every cleaned row must land in a valid bucket (completeness check)
SELECT COUNT(*) AS violations
FROM inspection
WHERE outcome_bucket NOT IN ('pass', 'fail', 'no_outcome')
   OR outcome_bucket IS NULL;

-- D': the four "didn't happen" raw statuses must map only to no_outcome,
-- and Fail/Pass/Pass w/ Conditions must map to fail/pass (mapping-consistency check)
SELECT COUNT(*) AS violations
FROM inspection
WHERE (raw_result IN ('Out of Business', 'No Entry', 'Not Ready', 'Business Not Located')
       AND outcome_bucket <> 'no_outcome')
   OR (raw_result = 'Fail' AND outcome_bucket <> 'fail')
   OR (raw_result IN ('Pass', 'Pass w/ Conditions') AND outcome_bucket <> 'pass');

================================================================================
IC5 - Referential integrity (no orphan rows after the join)
================================================================================
-- D': every non-ambiguous inspection must join to an establishment
-- (the 60 'ambiguous' rows are intentionally excluded - see notes/S5_change_summary.md)
SELECT COUNT(*) AS violations
FROM inspection i
LEFT JOIN establishment e ON e.license_no = i.license_no
WHERE e.license_no IS NULL
  AND i.license_fix_status <> 'ambiguous';

-- D': every violation_finding row must reference a real inspection
SELECT COUNT(*) AS violations
FROM violation_finding vf
LEFT JOIN inspection i ON i.inspection_id = vf.inspection_id
WHERE i.inspection_id IS NULL;

-- D': every violation_finding row must reference a defined violation_code
SELECT COUNT(*) AS violations
FROM violation_finding vf
LEFT JOIN violation_code vc ON vc.violation_code = vf.violation_code
WHERE vc.violation_code IS NULL;
